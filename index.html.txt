        document.getElementById("themeButton");

        document.body.classList.contains("dark-mode")
    ) {

        button.textContent = "☀";

        button.textContent = "☾";

/* =========================================================

const gridSize = 20;

const tileSize =
    canvas.width / gridSize;


/* =========================================================

let snake = [];

let direction = {

let nextDirection = {

/* =========================================================

/*

let foods = [];


/* =========================================================

let score = 0;

let highScore =
    Number(

            "nutritionSnakeHighScore"

let gameRunning = false;

let paused = false;

let gameOver = false;

let lastMoveTime = 0;

let moveDelay = 135;


/* =========================================================

const healthyFoods = [

/* =========================================================
   START GAME
========================================================= */

function startGame() {

    snake = [

        {
            x: 10,
            y: 10
        },

        {
            x: 9,
            y: 10
        },

        {
            x: 8,
            y: 10
        },

        {
            x: 7,
            y: 10
        }

    ];


    direction = {
        x: 1,
        y: 0
    };


    nextDirection = {
        x: 1,
        y: 0
    };


    score = 0;

    moveDelay = 135;

    gameRunning = true;

    paused = false;

    gameOver = false;


    scoreDisplay.textContent =
        score;


    overlay.classList.remove(
        "show"
    );


    /*
        Create the full collection
        of foods at the beginning.
    */

    foods = [];


    for (
        let i = 0;
        i < HEALTHY_COUNT;
        i++
    ) {

        spawnFood("healthy");

    }


    for (
        let i = 0;
        i < JUNK_COUNT;
        i++
    ) {

        spawnFood("junk");

    }


    for (
        let i = 0;
        i < WATER_COUNT;
        i++
    ) {

        spawnFood("water");

    }


    lastMoveTime =
        performance.now();


    requestAnimationFrame(
        gameLoop
    );


    drawGame();
}


/* =========================================================
   GAME LOOP
========================================================= */

function gameLoop(time) {

    if (!gameRunning) {

        drawGame();

        return;
    }


    if (
        !paused &&
        !gameOver
    ) {

        if (
            time - lastMoveTime >=
            moveDelay
        ) {

            moveSnake();

            lastMoveTime = time;

        }

    }


    drawGame();


    requestAnimationFrame(
        gameLoop
    );
}


/* =========================================================
   MOVE SNAKE
========================================================= */

function moveSnake() {

    direction =
        nextDirection;


    const head = {

        x:
            snake[0].x +
            direction.x,

        y:
            snake[0].y +
            direction.y

    };


    /* WALL */

    if (

        head.x < 0 ||

        head.x >= gridSize ||

        head.y < 0 ||

        head.y >= gridSize

    ) {

        endGame(
            "You hit the wall!"
        );

        return;
    }


    /* SELF */

    for (
        let i = 0;
        i < snake.length;
        i++
    ) {

        if (

            head.x ===
            snake[i].x &&

            head.y ===
            snake[i].y

        ) {

            endGame(
                "You ran into yourself!"
            );

            return;
        }
    }


    /* ADD HEAD */

    snake.unshift(head);


    /*
        Check whether the snake
        landed on ANY food.
    */

    let eatenFood = null;


    for (
        let i = 0;
        i < foods.length;
        i++
    ) {

        if (

            head.x ===
            foods[i].x &&

            head.y ===
            foods[i].y

        ) {

            eatenFood =
                foods[i];

            break;
        }
    }


    /* FOOD WAS EATEN */

    if (eatenFood) {

        eatFood(
            eatenFood
        );


        /*
            Remove only the food
            that was eaten.
        */

        foods =
            foods.filter(
                item =>
                    item !==
                    eatenFood
            );


        /*
            Replace the eaten
            food with another food
            of the same category.
        */

        spawnFood(
            eatenFood.type
        );


    } else {

        /*
            No food eaten,
            so remove the tail.
        */

        snake.pop();

    }

}


/* =========================================================
   EAT FOOD
========================================================= */

function eatFood(foodItem) {


    /* HEALTHY */

    if (
        foodItem.type ===
        "healthy"
    ) {

        score += 1;

        /*
            The snake does not
            lose its tail this move,
            so it grows by 1.
        */

    }


    /* WATER */

    else if (
        foodItem.type ===
        "water"
    ) {

        score += 2;


        /*
            Water gives an
            extra growth segment.
        */

        const tail =
            snake[
                snake.length - 1
            ];


        snake.push({

            x: tail.x,

            y: tail.y

        });

    }


    /* JUNK */

    else if (
        foodItem.type ===
        "junk"
    ) {

        score -= 2;


        /*
            The snake already
            grows by 1 because
            we didn't pop the tail.

            Remove 3 segments
            total so the net
            result is a shrink.
        */

        if (
            snake.length > 3
        ) {

            snake.pop();

        }

        if (
            snake.length > 3
        ) {

            snake.pop();

        }

        if (
            snake.length > 3
        ) {

            snake.pop();

        }

    }


    /* SCORE FLOOR */

    if (score < 0) {

        score = 0;

    }


    /* HIGH SCORE */

    if (
        score > highScore
    ) {

        highScore =
            score;


        localStorage.setItem(

            "nutritionSnakeHighScore",

            highScore

        );


        highScoreDisplay.textContent =
            highScore;

    }


    scoreDisplay.textContent =
        score;


    /*
        Speed increases
        every 5 points.
    */

    moveDelay =
        Math.max(

            65,

            135 -
            Math.floor(
                score / 5
            ) * 6

        );

}


/* =========================================================
   SPAWN FOOD
========================================================= */

function spawnFood(type) {

    let position;

    let attempts = 0;


    /*
        Try to find a position
        that isn't occupied.
    */

    while (
        attempts < 500
    ) {

        position = {

            x:
                Math.floor(
                    Math.random() *
                    gridSize
                ),

            y:
                Math.floor(
                    Math.random() *
                    gridSize
                )

        };


        let occupied = false;


        /* CHECK SNAKE */

        for (
            const segment
            of snake
        ) {

            if (

                segment.x ===
                position.x &&

                segment.y ===
                position.y

            ) {

                occupied = true;

                break;
            }

        }


        /* CHECK OTHER FOOD */

        if (!occupied) {

            for (
                const existingFood
                of foods
            ) {

                if (

                    existingFood.x ===
                    position.x &&

                    existingFood.y ===
                    position.y

                ) {

                    occupied = true;

                    break;
                }

            }

        }


        if (!occupied) {

            break;

        }


        attempts++;

    }


    /*
        Select the food
        appearance.
    */

    let foodData;


    if (
        type ===
        "healthy"
    ) {

        foodData =
            healthyFoods[
                Math.floor(
                    Math.random() *
                    healthyFoods.length
                )
            ];

    }

    else if (
        type ===
        "junk"
    ) {

        foodData =
            junkFoods[
                Math.floor(
                    Math.random() *
                    junkFoods.length
                )
            ];

    }

    else {

        foodData =
            waterFood;

    }


    foods.push({

        x:
            position.x,

        y:
            position.y,

        type:
            type,

        emoji:
            foodData.emoji,

        name:
            foodData.name

    });

}


/* =========================================================
   DRAW GAME
========================================================= */

function drawGame() {

    const styles =
        getComputedStyle(
            document.body
        );


    const background =
        styles.getPropertyValue(
            "--game-background"
        );


    const grid =
        styles.getPropertyValue(
            "--grid"
        );


    const snakeHead =
        styles.getPropertyValue(
            "--snake-head"
        );


    const snakeBody =
        styles.getPropertyValue(
            "--snake-body"
        );


    /* BACKGROUND */

    ctx.fillStyle =
        background;


    ctx.fillRect(

        0,

        0,

        canvas.width,

        canvas.height

    );


    /* GRID */

    ctx.strokeStyle =
        grid;

    ctx.lineWidth = 1;


    for (
        let x = 0;
        x <= gridSize;
        x++
    ) {

        ctx.beginPath();

        ctx.moveTo(

            x * tileSize,

            0

        );

        ctx.lineTo(

            x * tileSize,

            canvas.height

        );

        ctx.stroke();

    }


    for (
        let y = 0;
        y <= gridSize;
        y++
    ) {

        ctx.beginPath();

        ctx.moveTo(

            0,

            y * tileSize

        );

        ctx.lineTo(

            canvas.width,

            y * tileSize

        );

        ctx.stroke();

    }


    /* =========================
       DRAW FOOD
    ========================= */

    for (
        const food
        of foods
    ) {


        let outline;


        if (
            food.type ===
            "healthy"
        ) {

            outline =
                "#4f8f68";

        }

        else if (
            food.type ===
            "junk"
        ) {

            outline =
                "#d95757";

        }

        else {

            outline =
                "#4d9ed8";

        }


        const centerX =
            food.x *
            tileSize +
            tileSize / 2;


        const centerY =
            food.y *
            tileSize +
            tileSize / 2;


        /*
            Coloured circle
            behind the emoji.
        */

        ctx.fillStyle =
            outline;


        ctx.beginPath();


        ctx.arc(

            centerX,

            centerY,

            tileSize * 0.39,

            0,

            Math.PI * 2

        );


        ctx.fill();


        /*
            Slightly smaller
            inner circle.
        */

        ctx.fillStyle =
            background;


        ctx.beginPath();


        ctx.arc(

            centerX,

            centerY,

            tileSize * 0.32,

            0,

            Math.PI * 2

        );


        ctx.fill();


        /*
            Emoji
        */

        ctx.font =
            `${tileSize * 0.72}px Arial`;


        ctx.textAlign =
            "center";


        ctx.textBaseline =
            "middle";


        ctx.fillText(

            food.emoji,

            centerX,

            centerY + 1

        );

    }


    /* =========================
       DRAW SNAKE
    ========================= */

    snake.forEach(

        (segment,index) => {

            const padding = 2.5;


            const x =
                segment.x *
                tileSize +
                padding;


            const y =
                segment.y *
                tileSize +
                padding;


            const size =
                tileSize -
                padding * 2;


            ctx.fillStyle =
                index === 0
                    ? snakeHead
                    : snakeBody;


            roundRect(

                ctx,

                x,

                y,

                size,

                size,

                5

            );


            ctx.fill();


            /*
                Eyes
            */

            if (
                index === 0
            ) {

                drawSnakeEyes(

                    segment,

                    direction

                );

            }

        }

    );


    /* =========================
       PAUSE
    ========================= */

    if (

        paused &&

        gameRunning &&

        !gameOver

    ) {

        ctx.fillStyle =
            "rgba(0,0,0,0.55)";


        ctx.fillRect(

            0,

            0,

            canvas.width,

            canvas.height

        );


        ctx.fillStyle =
            "white";


        ctx.font =
            "bold 32px Arial";


        ctx.textAlign =
            "center";


        ctx.textBaseline =
            "middle";


        ctx.fillText(

            "PAUSED",

            canvas.width / 2,

            canvas.height / 2

        );

    }

}


/* =========================================================
   ROUNDED RECTANGLE
========================================================= */

function roundRect(

    context,

    x,

    y,

    width,

    height,

    radius

) {

    context.beginPath();


    context.moveTo(

        x + radius,

        y

    );


    context.lineTo(

        x + width - radius,

        y

    );


    context.quadraticCurveTo(

        x + width,

        y,

        x + width,

        y + radius

    );


    context.lineTo(

        x + width,

        y + height - radius

    );


    context.quadraticCurveTo(

        x + width,

        y + height,

        x + width - radius,

        y + height

    );


    context.lineTo(

        x + radius,

        y + height

    );


    context.quadraticCurveTo(

        x,

        y + height,

        x,

        y + height - radius

    );


    context.lineTo(

        x,

        y + radius

    );


    context.quadraticCurveTo(

        x,

        y,

        x + radius,

        y

    );


    context.closePath();

}


/* =========================================================
   SNAKE EYES
========================================================= */

function drawSnakeEyes(
    segment,
    dir
) {

    const centerX =
        segment.x *
        tileSize +
        tileSize / 2;


    const centerY =
        segment.y *
        tileSize +
        tileSize / 2;


    let eyeOffsetX = 0;

    let eyeOffsetY = 0;


    if (
        dir.x !== 0
    ) {

        eyeOffsetX =
            dir.x * 4;

    }

    else {

        eyeOffsetY =
            dir.y * 4;

    }


    const eye1X =
        centerX +
        eyeOffsetX -
        (
            dir.y !== 0
                ? 4
                : 0
        );


    const eye1Y =
        centerY +
        eyeOffsetY -
        (
            dir.x !== 0
                ? 4
                : 0
        );


    const eye2X =
        centerX +
        eyeOffsetX +
        (
            dir.y !== 0
                ? 4
                : 0
        );


    const eye2Y =
        centerY +
        eyeOffsetY +
        (
            dir.x !== 0
                ? 4
                : 0
        );


    ctx.fillStyle =
        "white";


    ctx.beginPath();

    ctx.arc(

        eye1X,

        eye1Y,

        2.3,

        0,

        Math.PI * 2

    );

    ctx.fill();


    ctx.beginPath();

    ctx.arc(

        eye2X,

        eye2Y,

        2.3,

        0,

        Math.PI * 2

    );

    ctx.fill();


    ctx.fillStyle =
        "#222";


    ctx.beginPath();

    ctx.arc(

        eye1X,

        eye1Y,

        1,

        0,

        Math.PI * 2

    );

    ctx.fill();


    ctx.beginPath();

    ctx.arc(

        eye2X,

        eye2Y,

        1,

        0,

        Math.PI * 2

    );

    ctx.fill();

}


/* =========================================================
   GAME OVER
========================================================= */

function endGame(reason) {

    gameRunning = false;

    gameOver = true;


    if (
        score > highScore
    ) {

        highScore =
            score;


        localStorage.setItem(

            "nutritionSnakeHighScore",

            highScore

        );

    }


    scoreDisplay.textContent =
        score;


    highScoreDisplay.textContent =
        highScore;


    overlayTitle.textContent =
        "Game Over!";


    overlayMessage.textContent =
        reason +
        " Final score: " +
        score +
        ".";


    overlay.classList.add(
        "show"
    );


    drawGame();

}


/* =========================================================
   PAUSE
========================================================= */

function togglePause() {

    if (
        !gameRunning ||
        gameOver
    ) {

        return;

    }


    paused =
        !paused;


    drawGame();

}


/* =========================================================
   KEYBOARD CONTROLS
========================================================= */

document.addEventListener(

    "keydown",

    function(event) {

        const key =
            event.key.toLowerCase();


        /* SPACE */

        if (
            key === " "
        ) {

            event.preventDefault();

            togglePause();

            return;

        }


        if (paused) {

            return;

        }


        /* UP */

        if (

            key === "arrowup" ||

            key === "w"

        ) {

            if (
                direction.y !== 1
            ) {

                nextDirection = {

                    x: 0,

                    y: -1

                };

            }

            event.preventDefault();

        }


        /* DOWN */

        else if (

            key === "arrowdown" ||

            key === "s"

        ) {

            if (
                direction.y !== -1
            ) {

                nextDirection = {

                    x: 0,

                    y: 1

                };

            }

            event.preventDefault();

        }


        /* LEFT */

        else if (

            key === "arrowleft" ||

            key === "a"

        ) {

            if (
                direction.x !== 1
            ) {

                nextDirection = {

                    x: -1,

                    y: 0

                };

            }

            event.preventDefault();

        }


        /* RIGHT */

        else if (

            key === "arrowright" ||

            key === "d"

        ) {

            if (
                direction.x !== -1
            ) {

                nextDirection = {

                    x: 1,

                    y: 0

                };

            }

            event.preventDefault();

        }

    }

);


/* =========================================================
   MOBILE SWIPE CONTROLS
========================================================= */

let touchStartX = 0;

let touchStartY = 0;


canvas.addEventListener(

    "touchstart",

    function(event) {

        const touch =
            event.changedTouches[0];


        touchStartX =
            touch.clientX;


        touchStartY =
            touch.clientY;

    },

    {
        passive: true
    }

);


canvas.addEventListener(

    "touchend",

    function(event) {

        if (paused) {

            return;

        }


        const touch =
            event.changedTouches[0];


        const deltaX =
            touch.clientX -
            touchStartX;


        const deltaY =
            touch.clientY -
            touchStartY;


        const minimumSwipe =
            25;


        if (

            Math.abs(deltaX) <
            minimumSwipe &&

            Math.abs(deltaY) <
            minimumSwipe

        ) {

            return;

        }


        /* HORIZONTAL */

        if (

            Math.abs(deltaX) >
            Math.abs(deltaY)

        ) {

            if (
                deltaX > 0
            ) {

                if (
                    direction.x !== -1
                ) {

                    nextDirection = {

                        x: 1,

                        y: 0

                    };

                }

            }

            else {

                if (
                    direction.x !== 1
                ) {

                    nextDirection = {

                        x: -1,

                        y: 0

                    };

                }

            }

        }


        /* VERTICAL */

        else {

            if (
                deltaY > 0
            ) {

                if (
                    direction.y !== -1
                ) {

                    nextDirection = {

                        x: 0,

                        y: 1

                    };

                }

            }

            else {

                if (
                    direction.y !== 1
                ) {

                    nextDirection = {

                        x: 0,

                        y: -1

                    };

                }

            }

        }

    },

    {
        passive: true
    }

);
